cmake_minimum_required(VERSION 3.21)
project(NeurOSInstaller VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Quick QuickControls2 Widgets)
qt_add_executable(neuros-installer src/app/main.cpp)
target_link_libraries(neuros-installer PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::Widgets)
target_compile_definitions(neuros-installer PRIVATE NEUROS_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
install(TARGETS neuros-installer RUNTIME DESTINATION bin)
install(DIRECTORY src/app/ DESTINATION share/neuros-installer/app FILES_MATCHING PATTERN "*.qml")
install(DIRECTORY src/branding/NeurOS DESTINATION share/neuros-installer/branding)
